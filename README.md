int X = 10;
int Y = 20;
Console.WriteLine($"Equation: {X} + {Y} = {X + Y:C}");
 النتيجه ظهرت ك  $30.00 لان تم اضاقه :c ود طريقه جاهزه لتنسيق الارقام وتكون بشكل مقروء من غير ما نكتب ده بشكل يدوي 
 يعني بدل ما اكتب 30 $ اقدر اكتبها 30:c 
 وده بيسهل طريقه كتابه الكود وبيكون مقروء بشكل افضل وكمان لاني ممكن احتاج اكتر من فورمات لنفس الرقم 
 والفورمات مش بس بيكون علامه العمله ممكن يكون تحويل لبيناري او هيكسا او رقم منسق بعلامات عشريه او ممكن عدد عشري ثابت الخ ....
 عندنا مثلا int x = 123456 نقدر نعمله اكتر من فورمات زي
 x:c => $123456
 x:N => 123,456
 x:f2 => 12,35
 x = 0.25
 x:p => 25.00 %
